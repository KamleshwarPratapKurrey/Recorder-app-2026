import AsyncStorage from "@react-native-async-storage/async-storage";
import Slider from "@react-native-community/slider";
import { Audio } from "expo-av";
import * as FileSystem from "expo-file-system/legacy";
import * as Sharing from "expo-sharing";
import React, { useEffect, useRef, useState } from "react";
import {
  Alert,
  Animated,
  FlatList,
  Keyboard,
  Modal,
  Platform,
  StatusBar,
  StyleSheet,
  Text,
  TextInput,
  TouchableOpacity,
  TouchableWithoutFeedback,
  View,
} from "react-native";
import { SafeAreaView } from "react-native-safe-area-context";

interface RecordingItem {
  id: string;
  uris: string[];
  name: string;
  duration: string;
  durationMillis: number;
}

const STORAGE_KEY = "@voice_recorder_recordings";

export default function HomeScreen() {
  const [recording, setRecording] = useState<Audio.Recording | null>(null);
  const [isRecording, setIsRecording] = useState<boolean>(false);
  const [isPaused, setIsPaused] = useState<boolean>(false);
  const [recordings, setRecordings] = useState<RecordingItem[]>([]);
  const [searchQuery, setSearchQuery] = useState<string>("");
  const [sound, setSound] = useState<Audio.Sound | null>(null);
  const [playingId, setPlayingId] = useState<string | null>(null);
  const [selectedIds, setSelectedIds] = useState<string[]>([]);
  const [appendingItemId, setAppendingItemId] = useState<string | null>(null);

  // Modal management tracking state variable
  const [activeMenuCard, setActiveMenuCard] = useState<RecordingItem | null>(
    null,
  );

  // Playback Telemetry Trackers
  const [playbackPosition, setPlaybackPosition] = useState<number>(0);
  const [playbackDuration, setPlaybackDuration] = useState<number>(0);
  const [isSeeking, setIsSeeking] = useState<boolean>(false);

  const currentTrackIndexRef = useRef<number>(0);
  const accumulatedTimeRef = useRef<number>(0);

  // Real-time Recording Timer State
  const [recordingMillis, setRecordingMillis] = useState<number>(0);
  const recordingTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);

  // Keyboard Dynamic Metrics & List Ref
  const [keyboardHeight, setKeyboardHeight] = useState<number>(0);
  const flatListRef = useRef<FlatList<RecordingItem>>(null);

  // Animation drivers for the waveform
  const waveAnim1 = useRef(new Animated.Value(1)).current;
  const waveAnim2 = useRef(new Animated.Value(1)).current;
  const waveAnim3 = useRef(new Animated.Value(1)).current;

  const isRecordingActive = isRecording || isPaused;

  useEffect(() => {
    async function loadRecordings() {
      try {
        const savedRecordings = await AsyncStorage.getItem(STORAGE_KEY);
        if (savedRecordings) {
          const parsed = JSON.parse(savedRecordings);
          const migrated = parsed.map((item: any) => ({
            id: item.id,
            name: item.name,
            duration: item.duration,
            durationMillis: item.durationMillis || 0,
            uris: item.uris ? item.uris : [item.uri],
          }));
          setRecordings(migrated);
        }
      } catch (error) {
        console.error("Failed to load recordings:", error);
      }
    }

    async function getPermission() {
      const permission = await Audio.requestPermissionsAsync();
      if (permission.status !== "granted") {
        Alert.alert("Permission Denied", "Microphone access is required.");
      }
    }

    getPermission();
    loadRecordings();

    const showEvent =
      Platform.OS === "ios" ? "keyboardWillShow" : "keyboardDidShow";
    const hideEvent =
      Platform.OS === "ios" ? "keyboardWillHide" : "keyboardDidHide";

    const showListener = Keyboard.addListener(showEvent, (e) =>
      setKeyboardHeight(e.endCoordinates.height),
    );
    const hideListener = Keyboard.addListener(hideEvent, () =>
      setKeyboardHeight(0),
    );

    return () => {
      showListener.remove();
      hideListener.remove();
      clearRecordingTimer();
      if (sound) sound.unloadAsync();
    };
  }, []);

  // Handle Waveform Animation Loop
  useEffect(() => {
    let animationLoop: Animated.CompositeAnimation;

    if (isRecording && !isPaused) {
      const createWaveSequence = (
        animVar: Animated.Value,
        duration: number,
      ) => {
        return Animated.loop(
          Animated.sequence([
            Animated.timing(animVar, {
              toValue: 2.5,
              duration,
              useNativeDriver: true,
            }),
            Animated.timing(animVar, {
              toValue: 0.8,
              duration,
              useNativeDriver: true,
            }),
          ]),
        );
      };

      animationLoop = Animated.parallel([
        createWaveSequence(waveAnim1, 400),
        createWaveSequence(waveAnim2, 550),
        createWaveSequence(waveAnim3, 470),
      ]);
      animationLoop.start();
    } else {
      waveAnim1.setValue(1);
      waveAnim2.setValue(1);
      waveAnim3.setValue(1);
    }

    return () => {
      if (animationLoop) animationLoop.stop();
    };
  }, [isRecording, isPaused]);

  const handleInputFocus = (index: number) => {
    setTimeout(() => {
      if (flatListRef.current) {
        flatListRef.current.scrollToIndex({
          index,
          viewPosition: 0.3,
          animated: true,
        });
      }
    }, 150);
  };

  const startRecordingTimer = () => {
    clearRecordingTimer();
    recordingTimerRef.current = setInterval(
      () => setRecordingMillis((prev) => prev + 1000),
      1000,
    );
  };

  const pauseRecordingTimer = () => {
    if (recordingTimerRef.current) {
      clearInterval(recordingTimerRef.current);
      recordingTimerRef.current = null;
    }
  };

  const clearRecordingTimer = () => {
    if (recordingTimerRef.current) {
      clearInterval(recordingTimerRef.current);
      recordingTimerRef.current = null;
    }
    setRecordingMillis(0);
  };

  const formatDuration = (millis: number): string => {
    if (isNaN(millis) || millis < 0) return "0:00";
    const minutes = Math.floor(millis / 60000);
    const seconds = Math.floor((millis % 60000) / 1000);
    return `${minutes}:${seconds < 10 ? "0" : ""}${seconds}`;
  };

  const getFormattedDateTime = (): string => {
    const now = new Date();
    return `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, "0")}-${String(now.getDate()).padStart(2, "0")}_${String(now.getHours()).padStart(2, "0")}-${String(now.getMinutes()).padStart(2, "0")}-${String(now.getSeconds()).padStart(2, "0")}`;
  };

  async function handleRecordPress() {
    try {
      if (isRecordingActive) {
        if (!isPaused) {
          await recording?.pauseAsync();
          setIsPaused(true);
          pauseRecordingTimer();
        } else {
          await recording?.startAsync();
          setIsPaused(false);
          recordingTimerRef.current = setInterval(
            () => setRecordingMillis((prev) => prev + 1000),
            1000,
          );
        }
        return;
      }

      if (sound) {
        await sound.unloadAsync();
        setPlayingId(null);
        setSound(null);
      }

      await Audio.setAudioModeAsync({
        allowsRecordingIOS: true,
        playsInSilentModeIOS: true,
      });
      const newRecording = new Audio.Recording();
      await newRecording.prepareToRecordAsync(
        Audio.RecordingOptionsPresets.HIGH_QUALITY,
      );

      setRecording(newRecording);
      setIsRecording(true);
      setIsPaused(false);
      setAppendingItemId(null);
      startRecordingTimer();
      await newRecording.startAsync();
    } catch (err) {
      console.error(err);
    }
  }

  async function handleAppendPress(item: RecordingItem) {
    setActiveMenuCard(null); // Dismiss modal popup sheet instantly
    if (isRecordingActive) {
      Alert.alert(
        "Recording Active",
        "Please close or complete your current active session first.",
      );
      return;
    }
    if (sound) {
      await sound.unloadAsync();
      setPlayingId(null);
      setSound(null);
    }

    try {
      await Audio.setAudioModeAsync({
        allowsRecordingIOS: true,
        playsInSilentModeIOS: true,
      });
      const newRecording = new Audio.Recording();
      await newRecording.prepareToRecordAsync(
        Audio.RecordingOptionsPresets.HIGH_QUALITY,
      );

      setRecording(newRecording);
      setAppendingItemId(item.id);
      setIsRecording(true);
      setIsPaused(false);
      setRecordingMillis(item.durationMillis);

      recordingTimerRef.current = setInterval(
        () => setRecordingMillis((prev) => prev + 1000),
        1000,
      );
      await newRecording.startAsync();
    } catch (error) {
      console.error(error);
    }
  }

  async function finishRecording() {
    if (!recording) return;

    try {
      await recording.stopAndUnloadAsync();
      await Audio.setAudioModeAsync({ allowsRecordingIOS: false });

      const uri = recording.getURI();
      const status = await recording.getStatusAsync();

      if (uri && status.durationMillis) {
        if (appendingItemId) {
          const updatedList = recordings.map((item) => {
            if (item.id === appendingItemId) {
              const totalMillis = item.durationMillis + status.durationMillis;
              return {
                ...item,
                uris: [...item.uris, uri],
                durationMillis: totalMillis,
                duration: formatDuration(totalMillis),
              };
            }
            return item;
          });
          setRecordings(updatedList);
          await AsyncStorage.setItem(STORAGE_KEY, JSON.stringify(updatedList));
        } else {
          const nextIndex = recordings.length + 1;
          const compoundName = `Rec#${nextIndex}_${getFormattedDateTime()}`;

          const newRecording: RecordingItem = {
            id: Date.now().toString(),
            uris: [uri],
            name: compoundName,
            duration: formatDuration(status.durationMillis),
            durationMillis: status.durationMillis,
          };

          const updatedList = [newRecording, ...recordings];
          setRecordings(updatedList);
          await AsyncStorage.setItem(STORAGE_KEY, JSON.stringify(updatedList));
        }
      }
    } catch (error) {
      console.error(error);
    } finally {
      setRecording(null);
      setIsRecording(false);
      setIsPaused(false);
      setAppendingItemId(null);
      clearRecordingTimer();
    }
  }

  async function playPlaylistSegment(item: RecordingItem, index: number) {
    if (index >= item.uris.length) {
      stopPlayback();
      return;
    }

    try {
      currentTrackIndexRef.current = index;
      const { sound: newSound } = await Audio.Sound.createAsync(
        { uri: item.uris[index] },
        { shouldPlay: true },
      );
      setSound(newSound);

      newSound.setOnPlaybackStatusUpdate((status: any) => {
        if (status.isLoaded) {
          if (!isSeeking)
            setPlaybackPosition(
              accumulatedTimeRef.current + status.positionMillis,
            );
          if (status.didJustFinish) {
            newSound.unloadAsync();
            accumulatedTimeRef.current += status.durationMillis || 0;
            playPlaylistSegment(item, index + 1);
          }
        }
      });
    } catch (error) {
      console.error(error);
    }
  }

  async function playRecording(item: RecordingItem) {
    setActiveMenuCard(null); // Dismiss modal popup sheet instantly
    try {
      if (sound) {
        await sound.unloadAsync();
        setPlaybackPosition(0);
        accumulatedTimeRef.current = 0;
        if (playingId === item.id) {
          setPlayingId(null);
          setSound(null);
          return;
        }
      }

      setPlaybackDuration(item.durationMillis);
      setPlayingId(item.id);
      accumulatedTimeRef.current = 0;
      await playPlaylistSegment(item, 0);
    } catch (error) {
      console.error(error);
    }
  }

  async function stopPlayback() {
    if (sound) {
      await sound.unloadAsync();
    }
    setPlayingId(null);
    setSound(null);
    setPlaybackPosition(0);
    accumulatedTimeRef.current = 0;
  }

  const handleSlidingStart = () => setIsSeeking(true);

  const handleSlidingComplete = async (value: number) => {
    if (!sound || !playingId) return;
    const targetItem = recordings.find((r) => r.id === playingId);
    if (!targetItem) return;

    let totalCheck = 0;
    let targetIndex = 0;
    let targetOffset = value;

    for (let i = 0; i < targetItem.uris.length; i++) {
      try {
        const checkSound = new Audio.Sound();
        const checkStatus = await checkSound.loadAsync({
          uri: targetItem.uris[i],
        });
        const dMillis = checkStatus.isLoaded
          ? checkStatus.durationMillis || 0
          : 0;
        await checkSound.unloadAsync();

        if (value >= totalCheck && value <= totalCheck + dMillis) {
          targetIndex = i;
          targetOffset = value - totalCheck;
          break;
        }
        totalCheck += dMillis;
      } catch (e) {
        console.error(e);
      }
    }

    try {
      await sound.unloadAsync();
      accumulatedTimeRef.current = totalCheck;
      setIsSeeking(false);
      currentTrackIndexRef.current = targetIndex;

      const { sound: newSound } = await Audio.Sound.createAsync(
        { uri: targetItem.uris[targetIndex] },
        { shouldPlay: true, positionMillis: targetOffset },
      );
      setSound(newSound);
      setPlaybackPosition(value);

      newSound.setOnPlaybackStatusUpdate((status: any) => {
        if (status.isLoaded) {
          if (!isSeeking)
            setPlaybackPosition(
              accumulatedTimeRef.current + status.positionMillis,
            );
          if (status.didJustFinish) {
            newSound.unloadAsync();
            accumulatedTimeRef.current += status.durationMillis || 0;
            playPlaylistSegment(targetItem, targetIndex + 1);
          }
        }
      });
    } catch (err) {
      console.error(err);
      setIsSeeking(false);
    }
  };

  const updateFileName = async (id: string, newName: string) => {
    const updatedList = recordings.map((item) =>
      item.id === id ? { ...item, name: newName } : item,
    );
    setRecordings(updatedList);
    await AsyncStorage.setItem(STORAGE_KEY, JSON.stringify(updatedList));
  };

  const toggleSelectRecording = (id: string) => {
    setSelectedIds((prevSelected) =>
      prevSelected.includes(id)
        ? prevSelected.filter((sId) => sId !== id)
        : [...prevSelected, id],
    );
  };

  const handleSelectAllToggle = () => {
    if (selectedIds.length === filteredRecordings.length) {
      setSelectedIds([]);
    } else {
      setSelectedIds(filteredRecordings.map((rec) => rec.id));
    }
  };

  async function deleteRecording(item: RecordingItem) {
    setActiveMenuCard(null); // Dismiss modal popup sheet instantly
    Alert.alert(
      "Delete Recording",
      `Are you sure you want to delete "${item.name}" permanently?`,
      [
        { text: "Cancel", style: "cancel" },
        {
          text: "Delete",
          style: "destructive",
          onPress: async () => {
            try {
              if (playingId === item.id) {
                await stopPlayback();
              }
              await Promise.all(
                item.uris.map(async (uri) => {
                  const fileInfo = await FileSystem.getInfoAsync(uri);
                  if (fileInfo.exists) await FileSystem.deleteAsync(uri);
                }),
              );
              const updatedList = recordings.filter((r) => r.id !== item.id);
              setRecordings(updatedList);
              setSelectedIds((prev) => prev.filter((id) => id !== item.id));
              await AsyncStorage.setItem(
                STORAGE_KEY,
                JSON.stringify(updatedList),
              );
            } catch (error) {
              console.error(error);
            }
          },
        },
      ],
    );
  }

  async function deleteSelectedRecordings() {
    if (selectedIds.length === 0) return;
    Alert.alert(
      "Delete Selected",
      `Are you sure you want to delete ${selectedIds.length} items?`,
      [
        { text: "Cancel", style: "cancel" },
        {
          text: "Delete All",
          style: "destructive",
          onPress: async () => {
            try {
              let isPlayingDeleted = false;
              const itemsToDelete = recordings.filter((r) =>
                selectedIds.includes(r.id),
              );

              await Promise.all(
                itemsToDelete.map(async (item) => {
                  if (playingId === item.id) isPlayingDeleted = true;
                  await Promise.all(
                    item.uris.map(async (uri) => {
                      const fileInfo = await FileSystem.getInfoAsync(uri);
                      if (fileInfo.exists) await FileSystem.deleteAsync(uri);
                    }),
                  );
                }),
              );

              if (isPlayingDeleted) {
                await stopPlayback();
              }

              const updatedList = recordings.filter(
                (r) => !selectedIds.includes(r.id),
              );
              setRecordings(updatedList);
              setSelectedIds([]);
            } catch (error) {
              console.error(error);
            }
          },
        },
      ],
    );
  }

  async function shareRecording(item: RecordingItem) {
    setActiveMenuCard(null); // Dismiss modal popup sheet instantly
    const isAvailable = await Sharing.isAvailableAsync();
    if (!isAvailable || item.uris.length === 0) return;

    let tempUri: string | null = null;
    try {
      const primaryUri = item.uris[0];
      const sourceInfo = await FileSystem.getInfoAsync(primaryUri);
      if (!sourceInfo.exists) return;

      const baseFileName = primaryUri.split("/").pop() || "recording.m4a";
      const fileExtension = baseFileName.includes(".")
        ? baseFileName.split(".").pop()
        : "m4a";
      const cleanFileName = item.name.replace(/[^a-zA-Z0-9_-]/g, "_");
      tempUri = `${FileSystem.cacheDirectory}${cleanFileName}_${Date.now()}.${fileExtension}`;

      await FileSystem.copyAsync({ from: primaryUri, to: tempUri });
      await Sharing.shareAsync(tempUri, { mimeType: "audio/x-m4a" });
    } catch (error) {
      console.error(error);
    } finally {
      if (tempUri) {
        const tempInfo = await FileSystem.getInfoAsync(tempUri);
        if (tempInfo.exists)
          await FileSystem.deleteAsync(tempUri, { idempotent: true });
      }
    }
  }

  const filteredRecordings = recordings.filter((rec) =>
    rec.name.toLowerCase().includes(searchQuery.toLowerCase()),
  );

  const isAllSelected =
    filteredRecordings.length > 0 &&
    selectedIds.length === filteredRecordings.length;
  const currentlyPlayingItem = recordings.find((r) => r.id === playingId);

  const renderRecordingCard = ({
    item,
    index,
  }: {
    item: RecordingItem;
    index: number;
  }) => {
    const isSelected = selectedIds.includes(item.id);

    return (
      <View style={styles.card}>
        <View style={styles.cardHeaderGrid}>
          <TouchableOpacity
            style={styles.cardCheckboxHitbox}
            onPress={() => toggleSelectRecording(item.id)}
            activeOpacity={0.6}
          >
            <View
              style={[
                styles.cardCheckboxOutline,
                isSelected && styles.cardCheckboxChecked,
              ]}
            >
              {isSelected && <View style={styles.cardCheckboxInnerDot} />}
            </View>
          </TouchableOpacity>

          <View style={styles.cardTitleBlock}>
            <TextInput
              style={styles.fileNameInput}
              value={item.name}
              onChangeText={(text) => updateFileName(item.id, text)}
              onFocus={() => handleInputFocus(index)}
              placeholder="Rename recording"
              placeholderTextColor="#7a7a7a"
            />
            <Text style={styles.durationText}>{item.duration}</Text>
          </View>

          {/* Native Popup Trigger Button */}
          <TouchableOpacity
            style={styles.menuTriggerBtn}
            onPress={() => setActiveMenuCard(item)}
            activeOpacity={0.6}
          >
            <Text style={styles.menuTriggerText}>⚙ Actions</Text>
          </TouchableOpacity>
        </View>
      </View>
    );
  };

  return (
    <SafeAreaView style={styles.container} edges={["top", "left", "right"]}>
      <StatusBar barStyle="dark-content" />

      {/* FIXED TOP HEADER */}
      <View style={styles.fixedHeaderContainer}>
        <Text style={styles.headerTitle}>Voice Recorder</Text>

        <View style={styles.searchContainer}>
          <TextInput
            style={styles.searchBar}
            placeholder="🔎 Search by name or date (e.g. 2026-05)..."
            placeholderTextColor="#A0A5AA"
            value={searchQuery}
            onChangeText={setSearchQuery}
          />
        </View>

        {filteredRecordings.length > 0 && (
          <View style={styles.utilityHeaderRow}>
            <TouchableOpacity
              style={styles.selectAllToggleBtn}
              onPress={handleSelectAllToggle}
              activeOpacity={0.7}
            >
              <View
                style={[
                  styles.checkboxOutline,
                  isAllSelected && styles.checkboxChecked,
                ]}
              >
                {isAllSelected && <View style={styles.checkboxInnerDot} />}
              </View>
              <Text style={styles.utilityHeaderText}>
                {isAllSelected ? "Deselect All" : "Select All"}
              </Text>
            </TouchableOpacity>

            {selectedIds.length > 0 && (
              <TouchableOpacity
                style={styles.bulkDeleteButton}
                onPress={deleteSelectedRecordings}
                activeOpacity={0.7}
              >
                <Text style={styles.bulkDeleteText}>
                  🗑 Delete Selected ({selectedIds.length})
                </Text>
              </TouchableOpacity>
            )}
          </View>
        )}
      </View>

      {/* SCROLLING RECORDINGS LIST */}
      {filteredRecordings.length === 0 ? (
        <View style={styles.emptyContainer}>
          <Text style={styles.emptyText}>
            {searchQuery
              ? "No matching recordings found."
              : "No recordings yet. Tap below to start!"}
          </Text>
        </View>
      ) : (
        <FlatList
          ref={flatListRef}
          data={filteredRecordings}
          renderItem={renderRecordingCard}
          keyExtractor={(item) => item.id}
          contentContainerStyle={[
            styles.listScrollContent,
            {
              paddingBottom: playingId
                ? 240
                : keyboardHeight > 0
                  ? keyboardHeight + 100
                  : 160,
            },
          ]}
          keyboardShouldPersistTaps="handled"
          keyboardDismissMode="on-drag"
        />
      )}

      {/* NEW NATIVE MODAL POPUP BOTTOM SHEET */}
      <Modal
        visible={activeMenuCard !== null}
        transparent={true}
        animationType="slide"
        onRequestClose={() => setActiveMenuCard(null)}
      >
        <TouchableWithoutFeedback onPress={() => setActiveMenuCard(null)}>
          <View style={styles.modalOverlayBackground}>
            <TouchableWithoutFeedback>
              <View style={styles.popupActionSheetContent}>
                <View style={styles.popupDragNotch} />
                <Text style={styles.popupMenuTitle} numberOfLines={1}>
                  Options for: {activeMenuCard?.name}
                </Text>

                {activeMenuCard && (
                  <View style={styles.modalButtonStack}>
                    <TouchableOpacity
                      style={[styles.popoverActionOption, styles.btnPlay]}
                      onPress={() => playRecording(activeMenuCard)}
                    >
                      <Text style={styles.popoverOptionText}>
                        {playingId === activeMenuCard.id
                          ? "✕ Stop Playback"
                          : "▶ Play Audio"}
                      </Text>
                    </TouchableOpacity>

                    <TouchableOpacity
                      style={[styles.popoverActionOption, styles.btnPlay]}
                      onPress={() => handleAppendPress(activeMenuCard)}
                    >
                      <Text style={styles.popoverOptionText}>
                        🎙+ Append Segment
                      </Text>
                    </TouchableOpacity>

                    <TouchableOpacity
                      style={[styles.popoverActionOption, styles.btnShare]}
                      onPress={() => shareRecording(activeMenuCard)}
                    >
                      <Text style={styles.popoverOptionText}>
                        🔗 Share WhatsApp
                      </Text>
                    </TouchableOpacity>

                    <TouchableOpacity
                      style={[styles.popoverActionOption, styles.btnDelete]}
                      onPress={() => deleteRecording(activeMenuCard)}
                    >
                      <Text style={styles.popoverOptionText}>
                        🗑 Delete Item
                      </Text>
                    </TouchableOpacity>

                    <TouchableOpacity
                      style={styles.popupCloseButton}
                      onPress={() => setActiveMenuCard(null)}
                    >
                      <Text style={styles.popupCloseBtnText}>Close Menu</Text>
                    </TouchableOpacity>
                  </View>
                )}
              </View>
            </TouchableWithoutFeedback>
          </View>
        </TouchableWithoutFeedback>
      </Modal>

      {/* STATIC FIXED CONTROL FOOTER PANEL */}
      <View
        style={[
          styles.footer,
          keyboardHeight > 0 && styles.footerKeyboardActive,
        ]}
      >
        {/* GLOBAL MUSIC PLAYER ELEMENT EXTENSION */}
        {playingId && currentlyPlayingItem ? (
          <View style={styles.globalPlayerSheet}>
            <View style={styles.globalPlayerHeader}>
              <Text style={styles.globalPlayingName} numberOfLines={1}>
                🎵 Playing: {currentlyPlayingItem.name}
              </Text>
              <TouchableOpacity
                style={styles.globalStopBtn}
                onPress={stopPlayback}
              >
                <Text style={styles.globalStopText}>✕ Stop</Text>
              </TouchableOpacity>
            </View>

            <View style={styles.timelineContainer}>
              <Slider
                style={styles.slider}
                minimumValue={0}
                maximumValue={playbackDuration}
                value={playbackPosition}
                minimumTrackTintColor="#007AFF"
                maximumTrackTintColor="#D3D3D3"
                thumbTintColor="#007AFF"
                onSlidingStart={handleSlidingStart}
                onSlidingComplete={handleSlidingComplete}
              />
              <View style={styles.timeLabelContainer}>
                <Text style={styles.timeText}>
                  {formatDuration(playbackPosition)}
                </Text>
                <Text style={styles.timeText}>
                  -
                  {formatDuration(
                    Math.max(0, playbackDuration - playbackPosition),
                  )}
                </Text>
              </View>
            </View>
          </View>
        ) : (
          /* HARDWARE RECORD CONTROLS MATRIX */
          <>
            {isRecordingActive && (
              <View style={styles.waveContainer}>
                <Animated.View
                  style={[
                    styles.waveBar,
                    { transform: [{ scaleY: waveAnim1 }] },
                    isPaused && styles.wavePaused,
                  ]}
                />
                <Animated.View
                  style={[
                    styles.waveBar,
                    styles.waveBarLong,
                    { transform: [{ scaleY: waveAnim2 }] },
                    isPaused && styles.wavePaused,
                  ]}
                />
                <Animated.View
                  style={[
                    styles.waveBar,
                    { transform: [{ scaleY: waveAnim3 }] },
                    isPaused && styles.wavePaused,
                  ]}
                />
                <Animated.View
                  style={[
                    styles.waveBar,
                    styles.waveBarLong,
                    { transform: [{ scaleY: waveAnim1 }] },
                    isPaused && styles.wavePaused,
                  ]}
                />
                <Animated.View
                  style={[
                    styles.waveBar,
                    { transform: [{ scaleY: waveAnim2 }] },
                    isPaused && styles.wavePaused,
                  ]}
                />
              </View>
            )}

            <View style={styles.recordingRow}>
              <TouchableOpacity
                style={[
                  styles.recordButton,
                  isRecordingActive && styles.recordingActive,
                ]}
                onPress={handleRecordPress}
                activeOpacity={0.8}
              >
                {isRecordingActive ? (
                  isPaused ? (
                    <Text style={styles.controlIconText}>▶</Text>
                  ) : (
                    <Text style={styles.liveTimerText}>
                      {formatDuration(recordingMillis)}
                    </Text>
                  )
                ) : (
                  <View style={styles.innerRecordCircle} />
                )}
              </TouchableOpacity>

              {isRecordingActive && (
                <TouchableOpacity
                  style={styles.finishButton}
                  onPress={finishRecording}
                  activeOpacity={0.8}
                >
                  <Text style={styles.finishButtonText}>✓ Finish</Text>
                </TouchableOpacity>
              )}
            </View>

            <Text style={styles.statusText}>
              {appendingItemId
                ? `Appending audio segment to list item...`
                : !isRecordingActive
                  ? "Tap to start recording"
                  : isPaused
                    ? "Recording paused. Tap to resume or click Finish"
                    : "Recording active... Tap button to pause"}
            </Text>
          </>
        )}
      </View>
    </SafeAreaView>
  );
}
